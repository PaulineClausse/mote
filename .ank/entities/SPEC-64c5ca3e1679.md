### 9.1 Canaux MCAP

Chaque canal numérote ses messages par un `seq` contigu à partir de 0, porté par l'enveloppe réseau et par le champ `sequence` du message MCAP.

| Canal | Schéma | Fréquence typique | Contenu |
|---|---|---|---|
| `/session/info` | `SessionInfo` | une fois | version de schéma, appareil, build OS, versions SDK, configuration |
| `/cam/left/frame`, `/cam/right/frame` | `ImageFrame` | selon la sélection | image encodée, pose par frame, métadonnées |
| `/cam/left/intrinsics`, `/cam/right/intrinsics` | `CameraIntrinsics` | une fois, ou à chaque changement | modèle et paramètres |
| `/pose/hmd` | `PoseSample` | cadence d'affichage | pose du casque |
| `/depth/left`, `/depth/right` | `DepthFrame` | 5 Hz | profondeur brute et descripteur |
| `/events` | `Event` | irrégulier | tracking, recentrage, rejets du live |
| `/stats` | `Stats` | 1 Hz | compteurs, débit, latences |
| `/session/end` | `SessionEnd` | une fois | dernier `seq` de chaque canal |

### 9.2 Esquisse du schéma protobuf (v1 brouillon)

```proto
syntax = "proto3";
package mote.v1;

// Convention des poses telles que fournies par la source (ADR-009).
// Le repère canonique OpenCV n'apparaît pas ici : il n'existe qu'à l'ingest.
enum PoseConvention {
  POSE_CONVENTION_UNSPECIFIED = 0;
  UNITY = 1;     // main gauche, Y haut, avant = +Z
  OPENXR = 2;    // main droite, Y haut, avant = -Z
}

// Base de temps du timestamp d'origine.
enum TimeDomain {
  TIME_DOMAIN_UNSPECIFIED = 0;
  MONOTONIC = 1;        // CLOCK_MONOTONIC
  BOOTTIME = 2;         // CLOCK_BOOTTIME
  XR_RUNTIME = 3;       // temps du runtime XR / OVR
  UNKNOWN_DOMAIN = 4;
  UNIX = 5;             // heure murale (ex. DateTime du Timestamp MRUK)
}

message Pose3 {
  double px = 1; double py = 2; double pz = 3;                 // position (m)
  double qx = 4; double qy = 5; double qz = 6; double qw = 7;  // rotation
}

message PoseSample {
  int64 t_ns = 1;              // instant de validité, horloge mote (9.3)
  int64 src_time_ns = 2;       // timestamp d'origine
  TimeDomain src_domain = 3;
  PoseConvention convention = 4;
  Pose3 world_from_hmd = 5;
  bool predicted = 6;          // pose prédite pour un instant futur
}

enum ImageCodec {
  IMAGE_CODEC_UNSPECIFIED = 0;
  JPEG = 1;
  RAW_RGBA = 2;                // sessions de référence, local uniquement
}

message ImageFrame {
  uint64 pair_id = 1;          // identique pour la gauche et la droite d'une paire
  int64 t_ns = 2;              // instant de capture, horloge mote
  int64 src_time_ns = 3;       // timestamp d'origine de la frame
  TimeDomain src_domain = 4;
  uint32 width = 5; uint32 height = 6;
  ImageCodec codec = 7;
  bytes data = 8;
  PoseConvention convention = 9;
  Pose3 world_from_camera = 10;      // pose par frame ; absente si non fournie
  optional int64 exposure_ns = 11;
  optional uint32 iso = 12;
  float blur_score = 13;
}

enum DistortionModel {
  DISTORTION_UNSPECIFIED = 0;
  PINHOLE = 1;
  OPENCV = 2;                  // k1 k2 p1 p2 (k3 optionnel)
}

message CameraIntrinsics {
  uint32 width = 1; uint32 height = 2;
  double fx = 3; double fy = 4; double cx = 5; double cy = 6;
  DistortionModel model = 7;
  repeated double dist = 8;
  PoseConvention convention = 9;
  Pose3 hmd_from_camera = 10;  // extrinsèques ; absentes si non fournies
}

enum DepthEncoding {
  DEPTH_ENCODING_UNSPECIFIED = 0;
  F32_RAW = 1;                 // tel que fourni par l'API, non linéarisé
  U16_MM = 2;                  // linéarisé, millimètres (réservé, voir 9.5)
}

message DepthFrame {
  int64 t_ns = 1;
  int64 src_time_ns = 2;
  TimeDomain src_domain = 3;
  uint32 width = 4; uint32 height = 5;
  DepthEncoding encoding = 6;
  bytes data = 7;              // non compressé : la compression est celle des chunks MCAP
  PoseConvention convention = 8;
  Pose3 world_from_depth = 9;  // pose de création
  float fov_left_tan = 10; float fov_right_tan = 11;
  float fov_top_tan = 12;  float fov_down_tan = 13;
  float near_z = 14; float far_z = 15;   // far_z peut valoir +infini
  float min_depth = 16; float max_depth = 17;
}

enum EventType {
  EVENT_TYPE_UNSPECIFIED = 0;
  TRACKING_LOST = 1;
  TRACKING_RECOVERED = 2;
  RECENTER = 3;                // l'origine du tracking a changé
  PAIR_DROPPED_BUSY = 4;       // paire perdue par saturation du core (NFR-02)
  PAIR_REJECTED_UNSYNC = 5;    // écart gauche/droite supérieur au seuil
  LIVE_DROPPED = 6;            // message retiré du flux live (reste dans le fichier)
  NET_DISCONNECTED = 7;
  NET_CONNECTED = 8;
}

message Event {
  int64 t_ns = 1;
  EventType type = 2;
  string detail = 3;
}

message Stats {
  int64 t_ns = 1;
  uint64 pairs_pushed = 2; uint64 pairs_kept = 3; uint64 pairs_dropped = 4;
  uint64 bytes_written = 5; uint64 bytes_sent = 6;
  float net_mbps = 7; float encode_ms_avg = 8; float live_queue_fill = 9;
}

message SessionInfo {
  uint32 schema_version = 1;
  string session_id = 2;
  int64 start_unix_ns = 3;
  int64 start_t_ns = 4;        // même instant sur l'horloge mote
  string device_model = 5;
  string os_build = 6;
  string sdk_versions = 7;     // JSON libre
  string host_kind = 8;        // "unity"
  string core_version = 9;
  string tracking_space = 10;  // espace de référence des poses
  string config_json = 11;
}

message ChannelEnd { string channel = 1; uint64 last_seq = 2; }
message SessionEnd { repeated ChannelEnd channels = 1; }

// ---- Transport (ADR-006). Non stocké dans le MCAP. ----

message Hello { string session_id = 1; string auth_token = 2; uint32 schema_version = 3; }

message Envelope {             // casque -> PC
  string channel = 1;
  uint64 seq = 2;
  oneof body {
    Hello hello = 10;
    SessionInfo session_info = 11;
    ImageFrame image = 12;
    CameraIntrinsics intrinsics = 13;
    PoseSample pose = 14;
    DepthFrame depth = 15;
    Event event = 16;
    Stats stats = 17;
    SessionEnd session_end = 18;
  }
}

message ResendRequest { string channel = 1; uint64 first_seq = 2; uint64 last_seq = 3; }

message Control {              // PC -> casque
  oneof body {
    ResendRequest resend = 1;
    // réservé v1.5 : carte de couverture issue de l'entraînement
  }
}
```

Les champs de profondeur reprennent ceux des descripteurs de SRC-1 (pose de création, tangentes de FOV, near/far, dimensions) `[SOURCE]`, complétés par `minDepth` et `maxDepth` du descripteur Unity `[DOC]`. Le descripteur Unity exprime le FOV en angles : l'hôte les convertit en tangentes.

En v1, `CameraIntrinsics.model` vaut toujours `PINHOLE` et `dist` est vide, MRUK ne fournissant pas de distorsion `[DOC]` ; `hmd_from_camera` est rempli à partir de `LensOffset`.

### 9.3 Modèle de temps

- **Horloge mote** : `t_ns` est le `CLOCK_MONOTONIC` du casque, en nanosecondes. `SessionInfo` enregistre la correspondance avec l'heure Unix au démarrage. `[DECISION]`
- **Règle de conservation** : chaque message porte `t_ns` et son timestamp d'origine avec sa base de temps (`src_time_ns`, `src_domain`). Aucune donnée d'origine n'est écrasée.
- **Qui convertit** : l'hôte, seul à connaître la base de temps de ses sources. Si une source ne peut pas être convertie de façon exacte, l'hôte renseigne `t_ns` au mieux et l'ingest sait, par `src_domain`, que ce `t_ns` est approximatif.
- **Timestamp des images** : MRUK le fournit sous forme de `DateTime` `[DOC]`, donc vraisemblablement une heure murale ; l'hôte le stocke tel quel (`src_domain = UNIX`) et en déduit `t_ns` par l'écart entre heure murale et `CLOCK_MONOTONIC` relevé au démarrage. L'horloge sous-jacente et l'instant visé dans l'exposition ne sont pas documentés (Q-11, SP-2). `[A VERIFIER]` La latence de capture annoncée est de 20 à 40 ms `[DOC]` : une image ne doit jamais être datée de son instant de réception.
- **Pose des images** : la pose de référence est la pose par frame fournie par MRUK (`world_from_camera`, pose au timestamp de l'image `[DOC]`). La synchronisation image/pose relève donc du système ; mote la mesure (SP-2, NFR-01) mais ne la reconstruit pas.
- **Profondeur** : datée par le `createTime` de son descripteur `[DOC]`.
- **Pose du casque** : le flux `/pose/hmd` sert à la couverture, au diagnostic et au repli. Les poses lues dans la boucle de rendu Unity sont prédites pour l'instant d'affichage `[CONNU]` : l'hôte les marque `predicted` et les date de l'instant visé, pas de l'instant de lecture.
- **Interpolation** : réalisée uniquement par l'ingest (lerp en translation, slerp en rotation), pour les images sans pose par frame ou pour comparer pose par frame et pose du casque. Le core n'interpole pas.
- **Validation** : SP-2.

### 9.4 Conventions de repère

- **Repère canonique (ingest, dataset, outils)** : monde main droite, métrique, d'origine et d'orientation héritées du tracking ; caméras en convention OpenCV/COLMAP (X droite, Y bas, Z avant), poses exprimées en `world_from_camera`. `[DECISION]` Raison : c'est la convention de COLMAP et de plusieurs modèles feed-forward (NoPoSplat la documente ainsi `[SOURCE]`).
- **Export nerfstudio** : caméras en convention OpenGL (X droite, Y haut, regard vers -Z). Conversion à l'export uniquement.
- **Depuis Unity** : monde main gauche, Y haut, caméra regardant +Z. Hypothèse de départ : miroir de l'axe Z appliqué au monde et à la caméra, `T_gl = S * T_unity * S` avec `S = diag(1,1,-1,1)`, puis passage OpenGL vers OpenCV par inversion des axes Y et Z de la caméra. `[HYPOTHESE]`
- **Poses MRUK** : rien ne garantit que les poses caméra de MRUK suivent les conventions du rig Unity ; les notes de migration de SRC-1 montrent qu'on ne peut pas présumer. `[SOURCE]`
- **Espace de référence** : l'hôte fournit toutes les poses dans l'espace de tracking, indépendamment de la position du rig dans la scène Unity, et le déclare dans `SessionInfo.tracking_space`.
- **Recentrage** : un recentrage en cours de session change l'origine. L'hôte émet un événement `RECENTER` ; l'ingest découpe la session en segments à chaque recentrage et, en v1, ne conserve que le plus long. `[HYPOTHESE]`
- **Règle** : aucune formule de conversion n'est acceptée sans le test de validation ci-dessous.
- **Test de validation obligatoire** : sur une capture courte avec une cible connue (mire, objet à coins identifiables), reprojeter un point 3D connu dans plusieurs images avec les poses converties et vérifier l'erreur de reprojection, visuellement et numériquement ; vérifier aussi dans un viewer que les frustums forment une trajectoire cohérente, orientée vers la scène.

### 9.5 Profondeur

- Supportée sur Quest 3 et 3S, portée minimale fiable d'environ 0,2 m. `[DOC]`
- Valeurs non linéaires : `nearZ` et `farZ` sont les plans d'une projection OpenGL et servent à retrouver la distance métrique. `farZ` peut être infini ; la projection a alors pour quadrant bas-droit `[[-1, -2*nearZ], [-1, 0]]`. `[DOC]`
- Formule de travail pour le cas infini, déduite de cette matrice en supposant des valeurs `d` normalisées dans [0, 1] : `distance = nearZ / (1 - d)`. `[HYPOTHESE]` Le format et la normalisation des valeurs ne sont pas documentés : à confirmer en SP-3 sur une distance connue. `[A VERIFIER]`
- Livrée en texture GPU uniquement (swapchain lisible, `RenderTexture` côté Unity) `[DOC]` : l'hôte doit la relire vers le CPU (readback asynchrone). Coût et latence à mesurer en SP-3. Activer la profondeur a un surcoût même sans lecture `[DOC]`.
- Résolution et cadence non documentées, à lire à l'exécution. `[A VERIFIER]`
- **Stockage** : brut (`F32_RAW`) avec descripteurs, conformément à ADR-009. Linéarisation à l'ingest. Compression par les chunks MCAP uniquement. `[DECISION]`
- L'encodage `U16_MM` (linéarisé à bord) est réservé : il ne sera activé que si SP-3 confirme la formule et si le débit de profondeur pose problème.
- La profondeur porte sa propre pose de création : sa rétroprojection ne dépend d'aucun alignement temporel avec les images.
- La profondeur et l'image RGB n'ont ni les mêmes intrinsèques ni la même pose. Le recalage est une étape explicite de l'ingest.

### 9.6 Versionnage

- `schema_version` dans `SessionInfo` et dans `Hello`. Règle d'évolution : ajout de champs uniquement ; tout changement cassant incrémente la version et fournit un convertisseur.
